    <script>
        // Data Init
        let transactions = JSON.parse(localStorage.getItem('transactions_apex_pure')) || [];
        let currentLang = localStorage.getItem('apex_lang') || 'ar';
        let currentCurr = localStorage.getItem('apex_curr') || 'MAD';
        let monthlyBudget = parseFloat(localStorage.getItem('apex_budget')) || 0;
        let appPin = localStorage.getItem('apex_pin') || null;
        let currentFilter = 'all';
        let selectedChartCategory = null;
        let expenseChart = null;
        let editingIndex = -1; let modalActionType = ''; let modalActionIndex = -1;

        const catKeysAll = ['food', 'transport', 'bills', 'shopping', 'salary', 'other'];
        const chartColors = ['#f43f5e', '#3b82f6', '#eab308', '#a855f7', '#10b981', '#64748b'];

        document.getElementById('tx-date').valueAsDate = new Date();
        document.getElementById('budget-input').value = monthlyBudget || '';

        // --- NEW: PIN Lock Logic ---
        if(appPin) {
            document.getElementById('pin-screen').classList.remove('hidden');
            document.getElementById('pin-screen').classList.add('flex');
            document.getElementById('btn-pin-toggle').textContent = currentLang === 'ar' ? 'إزالة' : 'Remove';
            document.getElementById('btn-pin-toggle').classList.replace('text-cyan-400', 'text-rose-400');
            document.getElementById('btn-pin-toggle').classList.replace('border-cyan-900', 'border-rose-900');
        }

        function checkPin() {
            const input = document.getElementById('pin-input');
            if(input.value.length === 4) {
                if(input.value === appPin) {
                    document.getElementById('pin-screen').classList.add('hidden');
                    document.getElementById('pin-screen').classList.remove('flex');
                    input.value = '';
                } else {
                    input.classList.add('shake-anim', 'border-rose-500', 'text-rose-500');
                    document.getElementById('pin-error').classList.remove('hidden');
                    if(navigator.vibrate) navigator.vibrate([100, 50, 100]);
                    setTimeout(() => {
                        input.classList.remove('shake-anim', 'border-rose-500', 'text-rose-500');
                        document.getElementById('pin-error').classList.add('hidden');
                        input.value = '';
                    }, 500);
                }
            }
        }

        function togglePinSetup() {
            if(appPin) {
                // Remove PIN
                if(confirm(translations[currentLang].removePinConfirm)) {
                    localStorage.removeItem('apex_pin');
                    appPin = null;
                    document.getElementById('btn-pin-toggle').textContent = translations[currentLang].btnSetup;
                    document.getElementById('btn-pin-toggle').classList.replace('text-rose-400', 'text-cyan-400');
                    document.getElementById('btn-pin-toggle').classList.replace('border-rose-900', 'border-cyan-900');
                }
            } else {
                // Set PIN
                const newPin = prompt(translations[currentLang].setPinPrompt);
                if(newPin && newPin.length === 4 && !isNaN(newPin)) {
                    localStorage.setItem('apex_pin', newPin);
                    appPin = newPin;
                    document.getElementById('btn-pin-toggle').textContent = translations[currentLang].btnRemove;
                    document.getElementById('btn-pin-toggle').classList.replace('text-cyan-400', 'text-rose-400');
                    document.getElementById('btn-pin-toggle').classList.replace('border-cyan-900', 'border-rose-900');
                    alert(translations[currentLang].pinSuccess);
                } else if (newPin) {
                    alert(translations[currentLang].pinError);
                }
            }
        }

        // --- NEW: Recurring Expenses Logic (Runs on Load) ---
        const currentMonthStr = new Date().toISOString().substring(0,7);
        const lastOpenedMonth = localStorage.getItem('apex_last_month');

        function importCSV(event) {
            const file = event.target.files[0]; if (!file) return; const reader = new FileReader();
            reader.onload = function(e) {
                const text = e.target.result; const lines = text.split('\n');
                if(lines.length > 1) {
                    for(let i=1; i<lines.length; i++) {
                        if(!lines[i].trim()) continue;
                        const parts = lines[i].split(/,(?=(?:(?:[^"]*"){2})*[^"]*$)/);
                        if(parts.length >= 5) {
                            const date = parts[0].trim(); const desc = parts[1].replace(/(^"|"$)/g, '').trim(); const cat = parts[2].trim(); const type = parts[3].trim(); const amount = parseFloat(parts[4].trim());
                            const isRecur = parts[5] ? (parts[5].trim().toLowerCase() === 'yes') : false;
                            if(date && desc && !isNaN(amount)) transactions.push({ date, description: desc, category: cat, type: type === 'expense' || type === 'مصروف (-)' ? 'expense' : 'income', amount, isRecurring: isRecur });
                        }
                    }
                    transactions.sort((a,b) => new Date(b.date) - new Date(a.date)); alert(translations[currentLang].importSuccess); populateFilterDropdown(); updateUI(); closeSettings();
                }
            }; reader.readAsText(file); event.target.value = '';
        }

                        <div onclick="toggleCatFilter('${catKeysAll[i]}')" class="flex justify-between items-center p-2 rounded-lg border cursor-pointer hover:bg-slate-800 transition-all ${activeClass}">
                            <div class="flex items-center gap-2 overflow-hidden">
                                <span class="w-2.5 h-2.5 rounded-full shrink-0" style="background-color: ${chartColors[i]}"></span>
                                <span class="text-[11px] text-slate-300 truncate">${labelsCat[i]}</span>
                            </div>
                            <div class="flex items-center gap-2 shrink-0 ml-1">
                                <span class="text-[11px] font-bold text-slate-200">${amt.toFixed(0)}</span>
                                <span class="text-[9px] text-slate-500 w-6 text-right">${pct}%</span>
                            </div>
                        </div>
                    `;
                }
            });
            legendDiv.innerHTML = legendHTML;

            if (expenseChart) { 
                expenseChart.data.labels = labelsCat; expenseChart.data.datasets[0].data = dataCat; expenseChart.update(); 
            } else {
                expenseChart = new Chart(ctxCat, { 
                    type: 'doughnut', data: { labels: labelsCat, datasets: [{ data: dataCat, backgroundColor: chartColors, borderWidth: 2, borderColor: '#020617', hoverOffset: 4 }] }, 
                    options: { responsive: true, maintainAspectRatio: false, cutout: '75%', plugins: { legend: { display: false } }, onClick: (event, elements) => { if (elements.length > 0) toggleCatFilter(catKeysAll[elements[0].index]); } } 
                });
            }
